# generowanie klucza
openssl genpkey -algorithm RSA -out crypto_example.key -pkeyopt rsa_keygen_bits:4096

# generowania prosby o certyfikat
openssl req -new -key crypto_example.key -out crypto_example.csr -config crypto_example.cnf

# generowanie certyfikatu
openssl x509 -req -in crypto_example.csr -CA ../CN_CA.crt -CAkey ../CN_CA.key -CAcreateserial -out crypto_example_ext.crt -days 365 -extfile crypto_example_ext.cnf -extensions crypto_example_ext

# Kopiuje certyfikat
cp ../CN_CA.crt /usr/local/share/ca-certificates/
